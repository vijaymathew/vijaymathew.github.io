Write code that keeps its promises with Nex

Nex is a small, expression-oriented language with contracts, classes and a friendly REPL, designed to make good engineering habits the path of least resistance.

▶ Take the tour ⬇ Install Why Nex?
wallet.nex 
class Wallet
  feature
    money: Real

    spend(amount: Real)
      require
        enough: amount <= money
      do
        this.money := money - amount
      ensure
        decreased: money = old money - amount
      end
end

let w := create Wallet.with_balance(10.2)
w.spend(20.0)
Error: Precondition violation: enough
// why nex

Small on purpose.

Classes and inheritance sit beside first-class functions, so you can pick the right tool for each problem.

counter.nex
class Counter
  create
    make(start: Integer) do
      this.value := start
    end
  feature
    value: Integer
    inc() do
      this.value := this.value + 1
    end
    current(): Integer do
      result := value
    end
end
REPL
> let c: Counter := create Counter.make(10)
> c.inc
> print(c.current)
11
animals.nex
class Animal
  feature
    name: String
    speak do print(name) end
  create
    named(name: String) do
      this.name := name
    end
end

class Dog
  inherit Animal
  feature
    speak do
      print(name + " says woof")
    end
end
REPL
> let a: Animal := create Animal.named("Ko")
> a.speak
Ko
> let d: Animal := create Dog.named("Ki")
> d.speak
Ki says woof
functions.nex
function double(n: Integer): Integer
do
  result := n * 2
end

-- anonymous functions
let inc := fn (n: Integer): Integer do
  result := n + 1
end
REPL
> print(double(5))
10
> print(inc(10))
11

Nex compiles to JVM bytecode and can call any Java library. Ship one jar that runs anywhere Java does.

line_buffer.nex
import java.lang.StringBuilder

class Line_Buffer
  create
    make() do
      with "java" do
        this.builder := create StringBuilder
      end
    end
  feature
    builder: Any
    append_line(s: String) do
      with "java" do
        builder.append(s)
        builder.append("\n")
      end
    end
end
notes

Quick install

Installs the nex command system-wide, with REPL, formatter, docs and compiler tools.

bash <(curl -fsSL https://raw.githubusercontent.com/vijaymathew/nex/main/bootstrap-install.sh) jvm --install-deps
Without sudo

User-local install under ~/.local.

bash <(curl -fsSL https://raw.githubusercontent.com/vijaymathew/nex/main/bootstrap-install.sh) jvm --prefix "$HOME/.local"
